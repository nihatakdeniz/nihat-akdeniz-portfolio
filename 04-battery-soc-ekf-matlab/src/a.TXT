% -------------------------------------------------------------------------
% LİTYUM-İYON BATARYA SOC TAHMİNİ İÇİN GENİŞLETİLMİŞ KALMAN FİLTRESİ (EKF)
% Kaynak Tez: Buğra ÜVEZ, 2024 (Kırıkkale Üniversitesi)
% Model: 2RC Eşdeğer Devre Modeli
% Optimizasyon: Genetik Algoritma (GA) ile belirlenen Q, R, P matrisleri
% -------------------------------------------------------------------------

clc; clear; close all;

%% 1. AYARLAR VE VERİ SEÇİMİ
use_real_data = false; % [true] ise 'battery_dataset.mat' kullanır, [false] ise sentetik veri üretir.

% Batarya Kapasitesi (Tez Tablo 5.1: Turnigy Graphene 5000mAh)
Q_n_Ah = 5.0;           
Q_n = Q_n_Ah * 3600;    % Amper-saniye (As) cinsinden kapasite

%% 2. VERİ YÜKLEME / OLUŞTURMA
if use_real_data
    % Gerçek veri kullanımı
    try
        load('battery_dataset.mat'); % İçinde t, I, V, soc_ref olmalı
        % Boyut kontrolü
        if length(t) ~= length(I)
            error('Zaman (t) ve Akım (I) vektör boyutları eşit olmalı.');
        end
        dt = t(2) - t(1); % Örnekleme zamanı
    catch
        error('battery_dataset.mat dosyası bulunamadı veya değişkenler eksik.');
    end
else
    % Sentetik Veri Üretimi (Darbeli Deşarj Profili)
    dt = 1;                 % 1 saniye örnekleme
    T_total = 3000;         % Toplam süre
    t = 0:dt:T_total;
    
    % Akım Profili: Belirli aralıklarla 1C (5A) deşarj darbesi
    I = zeros(size(t));
    I(100:600) = 5.0;       % 5A Deşarj
    I(1000:1500) = 2.5;     % 2.5A Deşarj
    I(2000:2500) = 5.0;     % 5A Deşarj
    
    % Referans (Gerçek) SOC Hesabı (Coulomb Counting ile sentetik referans)
    soc_ref = zeros(size(t));
    soc_ref(1) = 1.0; % %100 dolu başla
    for k = 1:length(t)-1
        soc_ref(k+1) = soc_ref(k) - (I(k) * dt / Q_n);
    end
    
    % Sentetik Voltaj (Gürültü eklenmiş model çıktısı)
    V = zeros(size(t));
    % Basit model ile voltaj üret (EKF'nin neyi takip edeceğini görmek için)
    % Not: EKF aşağıda bu veriyi tahmin etmeye çalışacak.
    for k = 1:length(t)
        v_ocv = ocv_from_soc(soc_ref(k));
        V(k) = v_ocv - I(k)*0.015 - 0.01*randn; % R0 yaklaşık 15mOhm + gürültü
    end
end

%% 3. BATARYA MODEL PARAMETRELERİ (2RC Modeli)
% Tezde Turnigy Graphene 5000mAh kullanılmıştır.
% R ve C değerleri tezde doğrudan tablo olarak verilmediği için 
% tipik Li-ion parametreleri veya Khanum et al. (2021) referans alınarak atanmıştır.
% Gerçek deneysel parametreleriniz varsa burayı güncelleyin.

R0 = 0.015;    % Ohmik direnç (Ohm)
R1 = 0.005;    % 1. RC direnci (Ohm)
C1 = 2000;     % 1. RC kapasitansı (Farad)
R2 = 0.008;    % 2. RC direnci (Ohm)
C2 = 5000;     % 2. RC kapasitansı (Farad)

% Zaman Sabitleri
tau1 = R1 * C1;
tau2 = R2 * C2;

%% 4. OPTİMİZE EDİLMİŞ EKF PARAMETRELERİ (Tez Tablo 5.2, 5.3, 5.4)
% Bu değerler Tezdeki GA (Genetik Algoritma) sonuçlarıdır. 
% MSE: 0.000022 (%99.98 iyileştirme)

% İşlem Gürültüsü Kovaryansı (Q) - Tablo 5.3 (GA Sonuçları)
% Yapı: [Q_soc, 0, 0; 0, Q_v1, 0; 0, 0, Q_v2]
q_soc = 1.955e-6;
q_v1  = 8.972e-5;
q_v2  = 8.856e-5;
Q = diag([q_soc, q_v1, q_v2]);

% Ölçüm Gürültüsü Kovaryansı (R) - Tablo 5.4 (GA Sonuçları)
R_noise = 8.461e-5; 

% Tahmin Hatası Kovaryansı (P) - Başlangıç - Tablo 5.2 (GA Sonuçları)
P = diag([4.388e-5, 2.406e-5, 2.419e-5]);

%% 5. EKF BAŞLATMA
% Durum Vektörü: x = [SOC; V1; V2]
x_hat = [1.0; 0; 0]; % Başlangıç tahmini (%100 SOC, 0V polarizasyon)

% Sonuçları saklamak için
len = length(t);
soc_est = zeros(1, len);
Vt_est = zeros(1, len);
soc_error = zeros(1, len);

%% 6. SİMÜLASYON DÖNGÜSÜ (EKF Algoritması)
fprintf('Simülasyon başlatılıyor... (Adım sayısı: %d)\n', len);

for k = 1:len
    
    Ik = I(k);     % Akım (Deşarj pozitif kabul edilirse formülde eksi kullanılır, burada deşarj=pozitif kabul edilmiştir)
    Vk_meas = V(k); % Ölçülen Voltaj
    
    % --- ADIM 1: TAHMİN (Time Update) ---
    
    % Durum Geçiş Matrisi (A) - Tez Denklem 5.12
    a1 = exp(-dt/tau1);
    a2 = exp(-dt/tau2);
    A = [1, 0, 0; 
         0, a1, 0; 
         0, 0, a2];
    
    % Giriş Matrisi (B) - Tez Denklem 5.14
    % Not: Tezde deşarj akımı I(k) pozitif ise SOC azalır, bu yüzden -dt/Qn kullanılır.
    b1 = R1 * (1 - a1);
    b2 = R2 * (1 - a2);
    B = [-dt/Q_n; 
          b1; 
          b2];
      
    % Durum Tahmini (x_k-)
    x_hat_minus = A * x_hat + B * Ik;
    
    % Hata Kovaryans Tahmini (P_k-)
    P_minus = A * P * A' + Q;
    
    % --- ADIM 2: DÜZELTME (Measurement Update) ---
    
    % Tahmin edilen SOC ve Voltaj
    soc_pred = x_hat_minus(1);
    v1_pred  = x_hat_minus(2);
    v2_pred  = x_hat_minus(3);
    
    % OCV ve Türevi (Polinom Yaklaşımı)
    [Voc_pred, dOCV_dSOC] = ocv_from_soc(soc_pred);
    
    % Tahmini Terminal Voltajı (Vt = OCV - V1 - V2 - I*R0)
    % Deşarj akımı I pozitif ise V terminal düşer.
    Vt_predicted = Voc_pred - v1_pred - v2_pred - (Ik * R0);
    
    % Jacobian (C Matrisi) - Tez Denklem 5.15
    % C = [dOCV/dSOC, -1, -1]
    C = [dOCV_dSOC, -1, -1];
    
    % Kalman Kazancı (K)
    S = C * P_minus * C' + R_noise;
    K = (P_minus * C') / S;
    
    % Ölçüm Farkı (Innovation)
    y_tilde = Vk_meas - Vt_predicted;
    
    % Durum Güncelleme
    x_hat = x_hat_minus + K * y_tilde;
    
    % Kovaryans Güncelleme
    P = (eye(3) - K * C) * P_minus;
    
    % --- KAYIT ---
    soc_est(k) = x_hat(1);
    Vt_est(k) = Vt_predicted;
    soc_error(k) = soc_ref(k) - soc_est(k);
end

%% 7. SONUÇLARIN GÖRSELLEŞTİRİLMESİ
figure('Name', 'EKF SOC Tahmin Sonuçları', 'Color', 'w', 'Position', [100, 100, 800, 600]);

subplot(3,1,1);
plot(t/60, V, 'k', 'LineWidth', 1); hold on;
plot(t/60, Vt_est, 'r--', 'LineWidth', 1.5);
title('Terminal Voltajı: Ölçülen vs Tahmin Edilen');
legend('Ölçülen (Real/Synth)', 'EKF Tahmini');
xlabel('Zaman (dk)'); ylabel('Voltaj (V)'); grid on;

subplot(3,1,2);
plot(t/60, soc_ref*100, 'b', 'LineWidth', 1.5); hold on;
plot(t/60, soc_est*100, 'r--', 'LineWidth', 1.5);
title(['SOC Tahmini (GA Optimize: MSE $\approx$ ', num2str(mean(soc_error.^2), '%.2e'), ')'], 'Interpreter', 'latex');
legend('Gerçek SOC (Ref)', 'EKF Tahmini');
xlabel('Zaman (dk)'); ylabel('SOC (%)'); grid on;

subplot(3,1,3);
plot(t/60, soc_error*100, 'k', 'LineWidth', 1);
title('SOC Tahmin Hatası');
xlabel('Zaman (dk)'); ylabel('Hata (%)'); grid on;

fprintf('Simülasyon tamamlandı. Ortalama Karesel Hata (MSE): %.2e\n', mean(soc_error.^2));


%% --- YARDIMCI FONKSİYONLAR ---

function [voc, docv_dsoc] = ocv_from_soc(soc)
    % Tezdeki Şekil 5.5 (SOC-OCV Eğrisi) için tipik bir Li-ion polinom modeli.
    % Gerçek deneysel eğri katsayıları tez metninde açıkça verilmemiştir, 
    % ancak 6. dereceden bir polinom literatürde standarttır.
    
    % SOC sınırlandırması (0 ile 1 arası)
    soc = max(0, min(1, soc));
    
    % Katsayılar (Tipik NMC/LCO Li-ion karakteristiği)
    % V = p1*s^6 + p2*s^5 + ... + p7
    p = [-0.906, 2.15, -0.6, 0.3, 0.1, 0.8, 3.2]; 
    % Not: Bu katsayılar temsilidir. Tezdeki eğriye tam uyum için 
    % "docv_dsoc" türevi ile birlikte deneysel veriden fit edilmelidir.
    
    % OCV Hesapla
    voc = polyval(p, soc);
    
    % Türev Hesapla (dOCV/dSOC) - Jacobian için gerekli
    p_der = polyder(p);
    docv_dsoc = polyval(p_der, soc);
end
